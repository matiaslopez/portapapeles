Funciones — declaración, definición, prototipos y modos de paso
Estructura básica

int suma(int a, int b);

// Definición

  return a + b;
}

Paso por valor — recibe una copia
void duplicar_mal(int x) {
  x = x * 2;  // modifica la copia
}             // el original NO cambia

int n = 5;
duplicar_mal(n);
// n sigue siendo 5
Paso por referencia — opera sobre el original
void duplicar(int& x) {
  x = x * 2;  // modifica el original
}

int n = 5;
duplicar(n);
// n = 10
Paso por puntero — equivalente
void duplicar_p(int* p) {
  *p = (*p) * 2;
}

int n = 5;
duplicar_p(&n);  // pasar dirección
// n = 10
const referencia — sin copiar, sin modificar
// Eficiente para datos grandes
void mostrar(const int& x) {
  cout << x;
  // x = 1; ← error de compilación
}

// Usando referencias de salida
void minmax(int arr[], int n,
            int& mn, int& mx) {
  mn = mx = arr[0];
  for (int i=1; i<n; i++) {
    if (arr[i]<mn) mn=arr[i];
    if (arr[i]>mx) mx=arr[i];
  }
}

Memoria dinámica — new / delete
Variable individual
// Alocar
int* p = new int;
*p = 42;

// Alocar e inicializar
int* p = new int(42);

// Liberar (obligatorio)
delete p;

Array dinámico

// Tamaño conocido en runtime

// Liberar (delete[] para arrays)

Struct en heap (nodo típico)
struct Nodo {
  int   dato;
  Nodo* sig;
};

Nodo* n = new Nodo;
n->dato = 5;
n->sig  = nullptr;

// Al terminar:
delete n;
n = nullptr;
Reglas críticas
newdebe tener sudelete
new T→delete p
new T[n]→delete[] p
Memory leak: olvidar el delete. La memoria queda ocupada hasta que termina el programa.

Doble free: llamar delete dos veces sobre el mismo puntero → comportamiento indefinido.

Dangling pointer: usar el puntero después de delete. Asignar nullptr después de liberar ayuda a detectarlo.

Struct con puntero propio (nodo)
struct Nodo {
  int   dato;
  Nodo* siguiente;  // apunta al mismo tipo
};

// Crear nodo en heap
Nodo* n = new Nodo;
n->dato      = 5;
n->siguiente = nullptr;

// Encadenar dos nodos
Nodo* a = new Nodo;  a->dato = 1;
Nodo* b = new Nodo;  b->dato = 2;
a->siguiente = b;
b->siguiente = nullptr;

// Recorrer
Nodo* cur = a;
while (cur != nullptr) {
  cout << cur->dato << " ";
  cur = cur->siguiente;
}

Organización en .h / .cpp
lista.h — interfaz pública
// Include guard:
// evita doble inclusión
#ifndef LISTA_H
#define LISTA_H

struct Nodo {
  int   dato;
  Nodo* sig;
};

// Solo declaraciones
void insertar(Nodo*& cab, int d);
void imprimir(Nodo* cab);
void liberar(Nodo*& cab);

#endif // LISTA_H
lista.cpp — implementación
#include "lista.h"
#include <iostream>
using namespace std;

void insertar(Nodo*& cab, int d) {
  Nodo* n = new Nodo;
  n->dato = d;
  n->sig  = cab;
  cab     = n;
}
void imprimir(Nodo* cab) {
  while (cab) {
    cout << cab->dato << " ";
    cab = cab->sig;
  }
}
void liberar(Nodo*& cab) {
  while (cab) {
    Nodo* tmp = cab;
    cab = cab->sig;
    delete tmp;
  }
}
main.cpp
#include "lista.h"
#include <iostream>
using namespace std;

int main() {
  Nodo* lista = nullptr;
  insertar(lista, 10);
  insertar(lista, 20);
  insertar(lista, 30);
  imprimir(lista);
  liberar(lista);
  return 0;
}
